import { createContext, useCallback, useContext, useEffect, useMemo, useState } from "react";
import type { ReactNode } from "react";
import { TOKEN_KEY } from "../services/api";
import { authService } from "../services/authService";
import type { RegisterPayload, User } from "../types";

interface AuthContextValue {
  user: User | null;
  loading: boolean;
  login: (email: string, password: string) => Promise<void>;
  register: (payload: RegisterPayload) => Promise<void>;
  logout: () => Promise<void>;
}

const AuthContext = createContext<AuthContextValue | undefined>(undefined);

export function AuthProvider({ children }: { children: ReactNode }) {
  const [user, setUser] = useState<User | null>(null);
  const [loading, setLoading] = useState(true);

  // Restore session on page load
  useEffect(() => {
    const token = localStorage.getItem(TOKEN_KEY);
    if (!token) {
      setLoading(false);
      return;
    }
    authService
      .me()
      .then(setUser)
      .catch(() => localStorage.removeItem(TOKEN_KEY))
      .finally(() => setLoading(false));
  }, []);

  // Token expired / rejected somewhere else in the app
  useEffect(() => {
    const onExpired = () => setUser(null);
    window.addEventListener("auth:expired", onExpired);
    return () => window.removeEventListener("auth:expired", onExpired);
  }, []);

  const login = useCallback(async (email: string, password: string) => {
    const { access_token } = await authService.login(email, password);
    localStorage.setItem(TOKEN_KEY, access_token);
    setUser(await authService.me());
  }, []);

  const register = useCallback(
    async (payload: RegisterPayload) => {
      await authService.register(payload);
      await login(payload.email, payload.password);
    },
    [login]
  );

  const logout = useCallback(async () => {
    try {
      await authService.logout();
    } catch {
      /* token may already be invalid; ignore */
    }
    localStorage.removeItem(TOKEN_KEY);
    setUser(null);
  }, []);

  const value = useMemo(
    () => ({ user, loading, login, register, logout }),
    [user, loading, login, register, logout]
  );

  return <AuthContext.Provider value={value}>{children}</AuthContext.Provider>;
}

export function useAuth(): AuthContextValue {
  const ctx = useContext(AuthContext);
  if (!ctx) throw new Error("useAuth must be used inside <AuthProvider>");
  return ctx;
}

import { useEffect, useRef, useState } from "react";
import { TOKEN_KEY, WS_URL } from "../services/api";
import type { OnlineUser, SessionEvent } from "../types";

export type ConnectionStatus =
  | "connecting"
  | "connected"
  | "disconnected";

/**
 * Connects to /ws/sessions/{id}, authenticates using
 * the first message, tracks presence, reconnects with
 * backoff, and forwards events to onEvent.
 *
 * Phase 15:
 * - Redirect handling when authentication expires.
 * - Stop reconnection when a session is deleted.
 */
export function useSessionSocket(
  sessionId: string | undefined,
  onEvent: (event: SessionEvent) => void
) {
  const [status, setStatus] =
    useState<ConnectionStatus>("connecting");

  const [online, setOnline] = useState<OnlineUser[]>([]);

  const onEventRef = useRef(onEvent);

  useEffect(() => {
    onEventRef.current = onEvent;
  });

  useEffect(() => {
    if (!sessionId) return;

    let ws: WebSocket | null = null;
    let closedByUs = false;
    let retries = 0;

    let retryTimer:
      | ReturnType<typeof setTimeout>
      | undefined;

    let pingTimer:
      | ReturnType<typeof setInterval>
      | undefined;

    const connect = () => {
      setStatus("connecting");

      const socket = new WebSocket(
        `${WS_URL}/ws/sessions/${sessionId}`
      );

      ws = socket;

      // --------------------------------------------------
      // Authenticate after the WebSocket opens
      // --------------------------------------------------

      socket.onopen = () => {
        socket.send(
          JSON.stringify({
            type: "AUTH",
            token: localStorage.getItem(TOKEN_KEY),
          })
        );
      };

      // --------------------------------------------------
      // Handle incoming events
      // --------------------------------------------------

      socket.onmessage = (msg) => {
        let event: SessionEvent;

        try {
          event = JSON.parse(msg.data);
        } catch {
          return;
        }

        switch (event.type) {
          case "CONNECTED":
            retries = 0;

            setStatus("connected");
            setOnline(event.payload.online_users ?? []);

            clearInterval(pingTimer);

            pingTimer = setInterval(() => {
              if (socket.readyState === WebSocket.OPEN) {
                socket.send(
                  JSON.stringify({ type: "PING" })
                );
              }
            }, 25000);

            break;

          case "PONG":
            return;

          case "USER_JOINED":
            setOnline((prev) =>
              prev.some(
                (u) =>
                  u.user_id === event.payload.user.user_id
              )
                ? prev
                : [...prev, event.payload.user]
            );
            break;

          case "USER_LEFT":
            setOnline((prev) =>
              prev.filter(
                (u) =>
                  u.user_id !== event.payload.user.user_id
              )
            );
            break;
        }

        onEventRef.current(event);
      };

      // --------------------------------------------------
      // Phase 15: Handle WebSocket disconnection
      // --------------------------------------------------

      socket.onclose = (e) => {
        // Ignore stale sockets from earlier effect runs.
        if (closedByUs) return;

        clearInterval(pingTimer);

        setStatus("disconnected");
        setOnline([]);

        // 4401: Invalid or expired JWT.
        // Tell the application's authentication handler
        // to return the user to the login page.
        if (e.code === 4401) {
          window.dispatchEvent(
            new Event("auth:expired")
          );
        }

        // 4401: Invalid/expired authentication token.
        // 4403: User does not have session permission.
        // 4404: Session was deleted.
        // Reconnecting cannot resolve these conditions.
        if (
          e.code === 4401 ||
          e.code === 4403 ||
          e.code === 4404
        ) {
          return;
        }

        // Retry temporary connection failures.
        retries += 1;

        retryTimer = setTimeout(
          connect,
          Math.min(
            1000 * 2 ** retries,
            15000
          )
        );
      };
    };

    // --------------------------------------------------
    // Initial connection
    // --------------------------------------------------

    connect();

    // --------------------------------------------------
    // Cleanup when session changes or component unmounts
    // --------------------------------------------------

    return () => {
      closedByUs = true;

      clearTimeout(retryTimer);
      clearInterval(pingTimer);

      ws?.close();
    };
  }, [sessionId]);

  return {
    status,
    online,
  };
}

import axios from "axios";

export const TOKEN_KEY = "copilothub_token";
export const API_URL = import.meta.env.VITE_API_URL ?? "http://localhost:8000";
export const WS_URL = API_URL.replace(/^http/, "ws");

export const api = axios.create({
  baseURL: API_URL,
  timeout: 15000,
});

api.interceptors.request.use((config) => {
  const token = localStorage.getItem(TOKEN_KEY);
  if (token) config.headers.Authorization = `Bearer ${token}`;
  return config;
});

api.interceptors.response.use(
  (response) => response,
  (error) => {
    const isLogin = error.config?.url?.includes("/auth/login");
    if (
      axios.isAxiosError(error) &&
      error.response?.status === 401 &&
      !isLogin &&
      localStorage.getItem(TOKEN_KEY)
    ) {
      localStorage.removeItem(TOKEN_KEY);
      window.dispatchEvent(new Event("auth:expired"));
    }
    return Promise.reject(error);
  }
);

/** Turns any axios/FastAPI error into a readable message. */
export function getErrorMessage(err: unknown): string {
  if (axios.isAxiosError(err)) {
    if (!err.response) return "Cannot reach the server. Is the backend running?";
    const detail = err.response.data?.detail;
    if (typeof detail === "string") return detail;
    if (Array.isArray(detail)) {
      return detail
        .map((d: { msg?: string }) => (d.msg ?? "Invalid input").replace(/^Value error, /, ""))
        .join(". ");
    }
    return `Request failed (${err.response.status})`;
  }
  return "Something went wrong";
}

import { api } from "./api";
import type {
  HealthResponse,
  RegisterPayload,
  TokenResponse,
  User,
} from "../types";

export const authService = {
  // Register a new user
  register: (payload: RegisterPayload) =>
    api
      .post<User>("/auth/register", payload)
      .then((r) => r.data),

  // Authenticate user and receive JWT token
  login: (email: string, password: string) =>
    api
      .post<TokenResponse>("/auth/login", {
        email,
        password,
      })
      .then((r) => r.data),

  // Get currently authenticated user
  me: () =>
    api
      .get<User>("/auth/me")
      .then((r) => r.data),

  // Logout current user
  logout: () =>
    api.post("/auth/logout"),

  // Check backend health
  health: () =>
    api
      .get<HealthResponse>("/health")
      .then((r) => r.data),

  // ========================================================
  // Phase 15: Change Password
  // ========================================================

  changePassword: (
    current_password: string,
    new_password: string
  ) =>
    api
      .post<TokenResponse>("/auth/change-password", {
        current_password,
        new_password,
      })
      .then((r) => r.data),
};

from fastapi import APIRouter, Depends, HTTPException, Request, status
from sqlalchemy.exc import IntegrityError
from sqlalchemy.orm import Session

from app.api.deps import get_current_user
from app.api.limits import client_ip, enforce, ip_limit, user_limit
from app.core.security import create_access_token, verify_password
from app.db.session import get_db
from app.models.user import User
from app.schemas.auth import ChangePasswordRequest, LoginRequest, MessageResponse, TokenResponse
from app.schemas.user import UserCreate, UserRead
from app.services import user_service

router = APIRouter(prefix="/auth", tags=["auth"])


@router.post(
    "/register", response_model=UserRead, status_code=status.HTTP_201_CREATED,
    dependencies=[Depends(ip_limit("register", 10, 60))],
)
def register(data: UserCreate, db: Session = Depends(get_db)) -> User:
    if user_service.get_by_email(db, data.email):
        raise HTTPException(status.HTTP_409_CONFLICT, "Email is already registered")
    if user_service.get_by_username(db, data.username):
        raise HTTPException(status.HTTP_409_CONFLICT, "Username is already taken")
    try:
        return user_service.create_user(db, data)
    except IntegrityError:  # race between check and insert
        db.rollback()
        raise HTTPException(status.HTTP_409_CONFLICT, "Email or username already exists")


@router.post("/login", response_model=TokenResponse)
def login(data: LoginRequest, request: Request, db: Session = Depends(get_db)) -> TokenResponse:
    enforce("login-ip", client_ip(request), 20, 60)  # slows down guessing from one machine...
    enforce("login-email", data.email.lower(), 8, 60)  # ...and guessing against one account
    user = user_service.authenticate(db, data.email, data.password)
    if user is None:
        raise HTTPException(
            status.HTTP_401_UNAUTHORIZED,
            "Incorrect email or password",
            headers={"WWW-Authenticate": "Bearer"},
        )
    token, expires_in = create_access_token(str(user.id))
    return TokenResponse(access_token=token, expires_in=expires_in)


@router.get("/me", response_model=UserRead)
def me(current_user: User = Depends(get_current_user)) -> User:
    return current_user


@router.post("/logout", response_model=MessageResponse)
def logout(_: User = Depends(get_current_user)) -> MessageResponse:
    # JWTs are stateless: the client deletes its token. To revoke every token at once, change the password.
    return MessageResponse(message="Logged out")


@router.post(
    "/change-password", response_model=TokenResponse,
    dependencies=[Depends(user_limit("change-password", 5, 60))],
)
def change_password(
    data: ChangePasswordRequest,
    user: User = Depends(get_current_user),
    db: Session = Depends(get_db),
) -> TokenResponse:
    """Changes the password and invalidates ALL older tokens. Returns a fresh token for this device."""
    if not verify_password(data.current_password, user.hashed_password):
        raise HTTPException(status.HTTP_400_BAD_REQUEST, "Current password is incorrect")
    if data.current_password == data.new_password:
        raise HTTPException(status.HTTP_400_BAD_REQUEST, "The new password must be different")
    user_service.set_password(db, user, data.new_password)
    token, expires_in = create_access_token(str(user.id))
    return TokenResponse(access_token=token, expires_in=expires_in)

async def _generate_in_background(session_id: uuid.UUID) -> None:
    try:
        brain, _ = await agent_runner.brain_factory()
        info = await report_service.generate_report(session_id, brain)
        with db_session.SessionLocal() as db:
            await event_service.publish(db, session_id, EventType.REPORT_SAVED, {"report": info})
    except Exception as exc:
        logger.exception("Report generation failed")
        with db_session.SessionLocal() as db:
            await event_service.publish(
                db, session_id, EventType.AGENT_MESSAGE,
                {"kind": "warning", "text": f"The report could not be generated ({exc.__class__.__name__})."},
            )


@router.post("/{session_id}/report/generate", response_model=MessageResponse, status_code=status.HTTP_202_ACCEPTED)
async def generate_report(
    ctx: SessionContext = CanGenerate, db: Session = Depends(get_db)
) -> MessageResponse:
    """Build a fresh final report from what the session has produced so far (runs in the background)."""
    sid = ctx.session.id
    done = db.scalar(select(func.count()).select_from(AgentAction).where(
        AgentAction.session_id == sid, AgentAction.action_type == "EVALUATE", AgentAction.success.is_(True))) or 0
    if done == 0:
        raise HTTPException(status.HTTP_409_CONFLICT, "Nothing to report yet: no step has been completed")
    if report_service.is_generating(sid):
        raise HTTPException(status.HTTP_409_CONFLICT, "A report is already being generated")
    existing = db.scalar(select(func.count()).select_from(Report).where(Report.session_id == sid, Report.kind == "FINAL")) or 0
    if existing >= get_settings().max_final_reports_per_session:
        raise HTTPException(status.HTTP_409_CONFLICT, f"A session can hold at most {get_settings().max_final_reports_per_session} final reports")
    report_service._generating.add(sid)  # claim it now so a double click cannot start two
    task = asyncio.create_task(_generate_in_background(sid))
    _tasks.add(task)
    task.add_done_callback(_tasks.discard)
    return MessageResponse(message="Generating the report. It will appear when it is ready.")
